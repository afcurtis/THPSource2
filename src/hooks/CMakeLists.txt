cmake_minimum_required(VERSION 3.20)

# 32-bit Windows hook harness: opentony.dll (native replacement hooks and the
# platform HAL) and the dinput.dll proxy that injects it. From Linux use
# `tony hooks build` (clang-cl + lld-link, cmake/toolchains/win32-clang-cl.cmake);
# on Windows any `-A Win32` MSVC generator works.
project(OpenTonyHooksWin32 LANGUAGES C CXX)

if(NOT WIN32)
    message(STATUS "OpenTonyHooksWin32 is Windows-only; no targets configured")
    return()
endif()
if(CMAKE_SIZEOF_VOID_P EQUAL 8)
    message(FATAL_ERROR "OpenTonyHooksWin32 must be configured for a 32-bit/Win32 target")
endif()

# SDL3 headers (`tony setup sdl3`, re/config/sdl3.yml). SDL3.dll itself is
# loaded at run time only when a [hal] service selects SDL.
set(OPENTONY_SDL3_INCLUDE "" CACHE PATH "SDL3 include directory (contains SDL3/SDL.h)")
if(NOT EXISTS "${OPENTONY_SDL3_INCLUDE}/SDL3/SDL.h")
    message(FATAL_ERROR "OPENTONY_SDL3_INCLUDE must contain SDL3/SDL.h (run `tony setup sdl3`)")
endif()

set(OPENTONY_HAL_DIR "${CMAKE_CURRENT_SOURCE_DIR}/../hal")

# native_hooks_generated.h is checked in and regenerated by `tony hooks
# generate` (it needs the knowledge base, not just the manifest);
# `tony hooks verify` rejects a stale copy.
add_library(opentony SHARED
    opentony_main.cpp
    hook_runtime.cpp
    selftest.cpp
    clock.cpp
    session_trace.cpp
    oracle_script.cpp
    native/vector_math.cpp
    native/overlay_open.cpp
    mod_overlay.cpp
    hal/hal_runtime.cpp
    hal/hal_clock.cpp
    hal/hal_input.cpp
    hal/site_trace.cpp
    hal/video_trace.cpp
    hal/video_shadow.cpp
    hal/video_mirror.cpp
    hal/video_renderer.cpp
    hal/video_hal.cpp
    hal/video_surface.cpp
    hal/movie_hal.cpp
    hal/audio_hal.cpp
    "${OPENTONY_HAL_DIR}/hal.cpp"
    "${OPENTONY_HAL_DIR}/clock_pacer.cpp"
    "${OPENTONY_HAL_DIR}/keymap.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_loader.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_clock.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_window.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_input.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_video_gl.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_video_gpu.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_audio.cpp"
    "${OPENTONY_HAL_DIR}/video.cpp"
    "${OPENTONY_HAL_DIR}/movie_video.cpp"
    "${OPENTONY_HAL_DIR}/audio.cpp"
    "${OPENTONY_HAL_DIR}/oracle/d3d7_stream.cpp"
    "${OPENTONY_HAL_DIR}/oracle/ddraw_surface.cpp"
    "${OPENTONY_HAL_DIR}/oracle/video_gpu.cpp"
)
target_include_directories(opentony PRIVATE
    "${CMAKE_CURRENT_SOURCE_DIR}"
    "${CMAKE_CURRENT_SOURCE_DIR}/../runtime"
)
target_include_directories(opentony SYSTEM PRIVATE "${OPENTONY_SDL3_INCLUDE}")
target_link_libraries(opentony PRIVATE user32)
target_compile_definitions(opentony PRIVATE OT_HAL_EXPORTS)

# The image oracle's HAL half (src/hooks/tools/video_replay.cpp): replays a
# recorded retail frame through the SDL video backend.
add_executable(opentony_video_replay
    tools/video_replay.cpp
    "${OPENTONY_HAL_DIR}/video.cpp"
    "${OPENTONY_HAL_DIR}/keymap.cpp"
    "${OPENTONY_HAL_DIR}/oracle/d3d7_stream.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_loader.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_window.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_input.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_video_gl.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_video_gpu.cpp"
    "${OPENTONY_HAL_DIR}/oracle/video_gpu.cpp"
)
target_include_directories(opentony_video_replay SYSTEM PRIVATE "${OPENTONY_SDL3_INCLUDE}")
target_link_libraries(opentony_video_replay PRIVATE user32)

# The GL backend's texture pack against a real context (src/hooks/tools/texture_pack_test.cpp, run by tests/test_texture_pack_backend.py under Wine and Xvfb).
add_executable(opentony_texture_pack_test
    tools/texture_pack_test.cpp
    "${OPENTONY_HAL_DIR}/video.cpp"
    "${OPENTONY_HAL_DIR}/keymap.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_loader.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_window.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_input.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_video_gl.cpp"
    "${OPENTONY_HAL_DIR}/sdl/sdl_video_gpu.cpp"
    "${OPENTONY_HAL_DIR}/oracle/video_gpu.cpp"
)
target_include_directories(opentony_texture_pack_test SYSTEM PRIVATE "${OPENTONY_SDL3_INCLUDE}")
target_link_libraries(opentony_texture_pack_test PRIVATE user32)

add_library(opentony_dinput SHARED dinput_proxy.cpp dinput.def)
set_target_properties(opentony_dinput PROPERTIES OUTPUT_NAME dinput)

foreach(target opentony opentony_dinput opentony_video_replay opentony_texture_pack_test)
    set_target_properties(${target} PROPERTIES PREFIX "")
    target_compile_definitions(${target} PRIVATE WIN32_LEAN_AND_MEAN NOMINMAX)
    set_property(TARGET ${target} PROPERTY MSVC_RUNTIME_LIBRARY "MultiThreaded")
    if(MSVC)
        # No C++ exceptions or RTTI: the VC6 CRT used by the clang-cl build has
        # no __CxxFrameHandler3. (CMake appends /EHsc after the toolchain flags.)
        target_compile_options(${target} PRIVATE /W4 /EHs-c- /GR-)
    endif()
endforeach()
